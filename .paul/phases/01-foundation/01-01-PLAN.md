  from pathlib import Path
  from jinja2 import Template, Environment, FileSystemLoader
  from lxml import etree
  from typing import Optional

  class XMLGenerator:
      """Genera XML de factura electrónica para SRI Ecuador."""

      def __init__(self, invoice: "Invoice"):
          self.invoice = invoice

      def render(self) -> str:
          """Genera XML firmado."""
          context = self._build_context()
          template = self._get_template()
          raw_xml = template.render(**context)
          return self._validate_against_xsd(raw_xml)

      def _build_context(self) -> dict:
          """Construye contexto para plantilla Jinja2."""
          return {
              "invoice": self.invoice,
              "company": self.invoice.company,
              "customer": self.invoice.customer,
              "lines": list(self.invoice.lines.all()),
              "taxes": self._compute_taxes(),
          }

      def _get_template(self) -> Template:
          """Carga plantilla desde templates/."""
          env = Environment(loader=FileSystemLoader(
              Path(__file__).parent.parent / "templates" / "facturacion_ec"
          ))
          return env.get_template("factura.xml.j2")

      def _validate_against_xsd(self, xml: str) -> str:
          """Valida XML contra XSD."""
          xsd_path = Path(__file__).parent.parent / "schemas" / "factura.xsd"
          schema_doc = etree.parse(str(xsd_path))
          schema = etree.XMLSchema(schema_doc)
          doc = etree.fromstring(xml.encode("utf-8"))
          schema.assertValid(doc)
          return etree.tostring(doc, pretty_print=True).decode()
```

  from cryptography.hazmat.primitives import hashes, serialization
  from cryptography.hazmat.primitives.asymmetric import padding
  from cryptography.hazmat.primitives.serialization import pkcs12
  from cryptography.x509 import load_der_x509_certificate
  from pathlib import Path
  from lxml import etree
  import base64

  class DigitalSigner:
      """Firma documentos XML con certificado X.509 (PKCS#12)."""

      def __init__(self, p12_path: str, password: str):
          with open(p12_path, "rb") as f:
              private_key, certificate, _ = pkcs12.load_key_and_certificates(
                  f.read(), password.encode()
              )
          self.private_key = private_key
          self.certificate = certificate

      def sign(self, xml_string: str) -> str:
          """Firma XML y retorna con nodo Signature."""
          # 1. Canonicalizar XML (C14N)
          doc = etree.fromstring(xml_string.encode())
          canonical_xml = etree.tostring(doc, method="c14n")

          # 2. Hash SHA256
          digest = hashes.Hash(hashes.SHA256())
          digest.update(canonical_xml)
          digest_value = digest.finalize()

          # 3. Firmar hash con private key
          signature = self.private_key.sign(
              digest_value,
              padding.PKCS1v15(),
              hashes.SHA256()
          )

          # 4. Insertar Signature node en XML
          # ... (insertar según esquema W3C XML-DSig)
          signed_xml = self._insert_signature(xml_string, signature, self.certificate)

          return signed_xml
```
